        RAW["Dữ Liệu Thô BDD10K (Read-Only 100%)
data/bdd10k/images/train/ (7.000 - 10.000 ảnh)"] --> FAST_DEDUP["FastDedup Engine (>360 FPS CPU)

        FAST_DEDUP --> DROP["Loại Bỏ 3.006 Khung Hình Trùng Lặp"]
        FAST_DEDUP --> UNIQUE_POOL["Kho Ảnh Sạch Duy Nhất (6.994 ảnh)
data/processed/unique_pool/"]
        
        UNIQUE_POOL --> PROFILER["Hồ Sơ Vật Lý 5D Không Cần Nhãn
(Định danh 5 lát cắt: Đêm, Mờ, Che khuất, Sương mù, Mật độ)"]
        
        PROFILER --> SPLIT["Phân Chia 4 Tập (Group-Aware Video Isolation)"]
        SPLIT --> TEST["Frozen Gold Test Set (fixed_test)
1.400 ảnh (20.02%) Đóng Băng Vĩnh Viễn"]
        SPLIT --> VAL["Fixed Validation Set (fixed_val)
700 ảnh (10.01%) Dùng Chung Early Stopping"]
        SPLIT --> SEED_V0["Tập Mầm Seed V0 (seed_v0)
500 ảnh (Cân Bằng 4 Lát Cắt Khó)"]
        SPLIT --> UNLABELED_POOL["Kho Mỏ Unlabeled Pool
4.394 ảnh (Dành Riêng Cho Khai Phá)"]

        SEED_V0 --> TRAIN_V0["Huấn Luyện Model Mầm V0 (YOLO11n Cfg-3)
AdamW, Cosine Annealing, Focal Loss"]
        TRAIN_V0 --> MODEL_V0["Model Mầm V0: yolo11n_v0_best.pt
(Đo trên Test Set: mAP50 = 20.23%)"]

        UNLABELED_POOL -.-> TRAIN_100_RANDOM["Huấn Luyện Baseline 100% Dữ Liệu Random (4.894 ảnh)"]
        TRAIN_100_RANDOM -.-> TARGET_100["Khóa Mốc Đối Chuẩn Duy Nhất: 100% Random Target
Target_mAP50 = 26.85%"]
    end

    %% =========================================================================
    %% KHỐI 2: VÒNG LẶP ACTIVE LEARNING TỔNG QUÁT (CHU KỲ k = 1 -> n)
    %% =========================================================================
    subgraph ACTIVE_LOOP ["2. VÒNG LẶP ACTIVE LEARNING TỔNG QUÁT (CHU KỲ k = 1 ➔ n)"]
        direction TB

        START_LOOP["Bắt Đầu Vòng k (k = 1, 2, ..., n)
• Tập huấn luyện cũ: D_train(k-1)
• Trọng số khởi đầu: Model V(k-1)"]

        START_LOOP --> MINING["Bước 1: Khai Phá Ảnh Khó Theo Công Thức Super-Fusion
S(x_i) = U_margin * sum(C_w * G_w) * Ego_Lane * Optical_Hazard
Submodular Facility Location (PCA 32D) tuyển chọn K ảnh khó nhất"]

        MINING --> HUMAN_ORACLE["Bước 2: Con Người Gán Nhãn Sạch 100% (Human Annotation)
• Loại bỏ hoàn toàn AI Pre-labeling để tránh ô nhiễm Confirmation Bias
• Chuyên gia kiểm toán & vẽ nhãn chuẩn vàng (Zero Label Noise)"]

        HUMAN_ORACLE --> NEW_BATCH["Lô K ảnh có nhãn sạch mới: Batch(k)"]

        NEW_BATCH --> CUMULATIVE_MERGE["Bước 3: Gộp Tích Lũy Vào Tập Huấn Luyện Vòng Trước
D_train(k) = D_train(k-1) ∪ Batch(k)
(Vật chất hóa Symlinks 0 KB dung lượng đĩa)"]

        CUMULATIVE_MERGE --> WARM_START_TRAIN["Bước 4: Tái Huấn Luyện Kế Thừa Model V(k-1) Trước Đó (Warm-Start)
Huấn luyện trên D_train(k) với init_weights = model_v(k-1)_best.pt
(Continuous Fine-Tuning 8 Epochs)"]

        WARM_START_TRAIN --> MODEL_VK["Sinh Ra Checkpoint Model V(k): model_v(k)_best.pt"]

        MODEL_VK --> EVAL_GOLD_TEST["Bước 5: Đánh Giá Trên Frozen Gold Test Set (1.400 ảnh)
Đo Macro mAP50 của Model V(k)"]

        EVAL_GOLD_TEST --> COMPARE_TARGET{"Bước 6: So Sánh Với Mốc Trần 100% Random (26.85%)"}

        COMPARE_TARGET -- "Chưa vượt mốc trần & k < n" --> NEXT_ROUND["Tăng k ➔ k + 1
Kế thừa Model V(k) làm xuất phát điểm"]
        NEXT_ROUND ==>|Tiếp tục vòng lặp| START_LOOP
    end

    %% =========================================================================
